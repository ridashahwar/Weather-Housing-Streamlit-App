# --- 4.1 Housing Affordability Prediction (StatCan Rents) ---
print("\n--- 4.1 Housing Affordability Prediction (StatCan Rents) ---")

# Define features (X) and target (y)
# 'VALUE' is the average rent
# Features include weather, time, city, and mortgage rate.
# Ensuring all features are present and handling NaNs.
features_rent = [
    'Avg_Max_Temp', 'Avg_Min_Temp', 'Total_Precip_mm', 'Total_Snow_cm',
    'year', 'month', 'city', 'mortgage_rate'
]
target_rent = 'VALUE'

# Filter out rows with NaN in target or essential features
df_rent_model = df_combined.dropna(subset=[target_rent] + features_rent).copy()

if not df_rent_model.empty:
    X_rent = df_rent_model[features_rent]
    y_rent = df_rent_model[target_rent]

    # Feature Engineering: Cyclical encoding for month
    # This helps capture the cyclical nature of months without treating them as ordinal.
    X_rent['month_sin'] = np.sin(2 * np.pi * X_rent['month'] / 12)
    X_rent['month_cos'] = np.cos(2 * np.pi * X_rent['month'] / 12)
    X_rent = X_rent.drop('month', axis=1) # Drop original month column

    # Identify categorical and numerical features for preprocessing
    numerical_features_rent = ['Avg_Max_Temp', 'Avg_Min_Temp', 'Total_Precip_mm', 'Total_Snow_cm',
                               'year', 'mortgage_rate', 'month_sin', 'month_cos']
    categorical_features_rent = ['city']

    # Create a column transformer for preprocessing:
    # Scale numerical features, One-Hot Encode categorical features.
    preprocessor_rent = ColumnTransformer(
        transformers=[
            ('num', StandardScaler(), numerical_features_rent),
            ('cat', OneHotEncoder(handle_unknown='ignore'), categorical_features_rent)
        ])

    # Define models for regression
    models_rent = {
        'Linear Regression': LinearRegression(),
        'Random Forest Regressor': RandomForestRegressor(random_state=42)
    }

    # Split data into training and testing sets
    X_train_rent, X_test_rent, y_train_rent, y_test_rent = train_test_split(
        X_rent, y_rent, test_size=0.2, random_state=42
    )

    for name, model in models_rent.items():
        # Create a pipeline to streamline preprocessing and model training
        pipeline_rent = Pipeline(steps=[('preprocessor', preprocessor_rent),
                                        ('regressor', model)])
        
        print(f"\n--- Training {name} for Housing Affordability ---")
        pipeline_rent.fit(X_train_rent, y_train_rent)
        y_pred_rent = pipeline_rent.predict(X_test_rent)

        # Evaluate model performance
        rmse_rent = np.sqrt(mean_squared_error(y_test_rent, y_pred_rent))
        r2_rent = r2_score(y_test_rent, y_pred_rent)

        print(f"  RMSE: {rmse_rent:.2f}")
        print(f"  R-squared: {r2_rent:.2f}")
        print(f"  Interpretation: {r2_rent*100:.0f}% of the variance in Average Rent is explained by the model. (Note: This is for the overall model, not specific to smoothed plots.)")

        if name == 'Linear Regression':
            # Analyze coefficients for Linear Regression
            # Get feature names after one-hot encoding for categorical features
            ohe_feature_names = pipeline_rent.named_steps['preprocessor'].named_transformers_['cat'].get_feature_names_out(categorical_features_rent)
            all_feature_names = numerical_features_rent + list(ohe_feature_names)
            coefficients = pd.Series(pipeline_rent.named_steps['regressor'].coef_, index=all_feature_names)
            print("\n  Linear Regression Coefficients (Top/Bottom 5):")
            print(coefficients.sort_values(ascending=False).head(5))
            print(coefficients.sort_values(ascending=True).head(5))
        elif name == 'Random Forest Regressor':
            # Analyze feature importance for Random Forest
            feature_importances = pipeline_rent.named_steps['regressor'].feature_importances_
            ohe_feature_names = pipeline_rent.named_steps['preprocessor'].named_transformers_['cat'].get_feature_names_out(categorical_features_rent)
            all_feature_names = numerical_features_rent + list(ohe_feature_names)
            importances_df = pd.DataFrame({'feature': all_feature_names, 'importance': feature_importances})
            print("\n  Random Forest Feature Importances (Top 10):")
            print(importances_df.sort_values('importance', ascending=False).head(10))
    
    print("\n  Considerations for Housing Affordability Prediction:")
    print("  - R-squared is moderate, suggesting other factors (e.g., local economy, vacancy rates) influence rent.")
    print("  - Explore adding more socio-economic features or time-series specific models for better performance.")
    print("  - Use cross-validation for more robust evaluation.")
else:
    print("Not enough data for Housing Affordability Prediction after dropping NaNs.")

# --- 4.3 Rental Housing Quality Prediction (Dwelling Condition/Damage - HES Data & Housing Indicators) ---
print("\n--- 4.3 Rental Housing Quality Prediction ---")

# --- Sub-problem 1: dwelling_damage (Binary Classification) ---
print("\n--- Sub-problem 1: Dwelling Damage Prediction (Binary Classification) ---")

# Filter for HES data (mostly 2023) and relevant columns
df_damage_model = df_combined.dropna(subset=['avg_dwelling_damage_percent']).copy()

if not df_damage_model.empty:
    # Create binary target: 1 if damage > 0, 0 otherwise
    df_damage_model['has_dwelling_damage'] = (df_damage_model['avg_dwelling_damage_percent'] > 0).astype(int)

    features_damage = [
        'Total_Precip_mm', 'Total_Snow_cm', 'Avg_Max_Temp', 'Avg_Min_Temp',
        'year', 'city'
    ]
    target_damage = 'has_dwelling_damage'

    X_damage = df_damage_model[features_damage]
    y_damage = df_damage_model[target_damage]

    # Identify numerical and categorical features
    numerical_features_damage = ['Total_Precip_mm', 'Total_Snow_cm', 'Avg_Max_Temp', 'Avg_Min_Temp', 'year']
    categorical_features_damage = ['city']

    preprocessor_damage = ColumnTransformer(
        transformers=[
            ('num', StandardScaler(), numerical_features_damage),
            ('cat', OneHotEncoder(handle_unknown='ignore'), categorical_features_damage)
        ])

    models_damage = {
        'Logistic Regression': LogisticRegression(random_state=42, solver='liblinear'),
        'Gradient Boosting Classifier': GradientBoostingClassifier(random_state=42)
    }

    X_train_damage, X_test_damage, y_train_damage, y_test_damage = train_test_split(
        X_damage, y_damage, test_size=0.2, random_state=42, stratify=y_damage # Stratify for imbalance
    )

    # Apply SMOTE to training data for class imbalance
    # Only apply if there's a minority class (i.e., both classes are present)
    if y_train_damage.value_counts().min() > 0:
        smote = SMOTE(random_state=42)
        # Preprocess training data BEFORE SMOTE
        X_train_damage_processed = preprocessor_damage.fit_transform(X_train_damage)
        X_train_damage_resampled, y_train_damage_resampled = smote.fit_resample(
            X_train_damage_processed, y_train_damage
        )
        print(f"\n  Original training class distribution: {y_train_damage.value_counts()}")
        print(f"  Resampled training class distribution: {y_train_damage_resampled.value_counts()}")
    else:
        # If only one class is present (e.g., all 0s or all 1s), SMOTE cannot be applied.
        X_train_damage_resampled = preprocessor_damage.fit_transform(X_train_damage)
        y_train_damage_resampled = y_damage
        print("\n  SMOTE not applied: Only one class present in training data or min class count is 0.")


    # Transform test data using the *fitted* preprocessor (do not resample test data)
    X_test_damage_processed = preprocessor_damage.transform(X_test_damage)

    for name, model in models_damage.items():
        print(f"\n--- Training {name} for Dwelling Damage ---")
        model.fit(X_train_damage_resampled, y_train_damage_resampled)
        y_pred_damage = model.predict(X_test_damage_processed)

        print(f"  Classification Report for {name}:\n{classification_report(y_test_damage, y_pred_damage)}")
        print(f"  Confusion Matrix for {name}:\n{confusion_matrix(y_test_damage, y_pred_damage)}")
    
    print("\n  Considerations for Dwelling Damage Prediction:")
    print("  - Models show strong performance (high accuracy, precision, recall).")
    print("  - The 60 False Negatives (actual damage predicted as no damage) might be critical depending on the cost of missing damage.")
    print("  - Explore additional features like property age, construction type, or specific hazard zones if available.")
    print("  - Use cross-validation for more robust evaluation.")
else:
    print("Not enough data for Dwelling Damage Prediction after dropping NaNs.")

# --- 4.5 Geographic Vulnerability (Clustering) ---
print("\n--- 4.5 Geographic Vulnerability (Clustering) ---")

# Aggregate city-level average values across all available years
# Ensure to drop NaNs before aggregation if a feature is critical for clustering
df_clustering = df_combined.groupby('city').agg(
    Avg_Rent=('VALUE', 'mean'),
    Avg_House_Price=('avg_house_price', 'mean'),
    Avg_Total_Precip_mm=('Total_Precip_mm', 'mean'),
    Avg_Total_Snow_cm=('Total_Snow_cm', 'mean'),
    Avg_Max_Temp=('Avg_Max_Temp', 'mean'),
    Avg_Min_Temp=('Avg_Min_Temp', 'mean')
).reset_index()

# Drop rows where any of the aggregated features are NaN (e.g., if a city has no house price data)
df_clustering.dropna(inplace=True)

if not df_clustering.empty:
    X_cluster = df_clustering.drop('city', axis=1)

    # Scale the features for clustering
    scaler_cluster = StandardScaler()
    X_scaled_cluster = scaler_cluster.fit_transform(X_cluster)

    # Determine optimal number of clusters (Elbow Method - optional but good practice)
    # Using a range of K from 1 to min(len(df_clustering) - 1, 5)
    max_k = min(len(df_clustering) - 1, 5) if len(df_clustering) > 1 else 1
    if max_k > 0:
        inertias = []
        k_range = range(1, max_k + 1)
        for k in k_range:
            kmeans = KMeans(n_clusters=k, random_state=42, n_init=10)
            kmeans.fit(X_scaled_cluster)
            inertias.append(kmeans.inertia_)

        plt.figure(figsize=(10, 6))
        plt.plot(k_range, inertias, marker='o')
        plt.title('Elbow Method for Optimal K')
        plt.xlabel('Number of Clusters (K)')
        plt.ylabel('Inertia')
        plt.xticks(k_range)
        plt.grid(True, linestyle='--', alpha=0.6)
        plt.show()

        # Choose a K (e.g., 3, or based on elbow method if clear)
        # Given only 3 cities, K=3 will simply assign each city to its own cluster.
        # If more cities were present, the elbow method would be more informative.
        n_clusters = 3 if len(df_clustering) >= 3 else len(df_clustering) if len(df_clustering) > 0 else 1
        if n_clusters > 0:
            kmeans = KMeans(n_clusters=n_clusters, random_state=42, n_init=10)
            df_clustering['Cluster'] = kmeans.fit_predict(X_scaled_cluster)

            print(f"\n  KMeans Clustering Results (K={n_clusters}):")
            print(df_clustering[['city', 'Cluster']])

            # Visualize clusters (using two key features, or PCA if many features)
            # For simplicity, let's pick Avg_Rent and Avg_Max_Temp for visualization
            plt.figure(figsize=(10, 8))
            sns.scatterplot(data=df_clustering, x='Avg_Rent', y='Avg_Max_Temp', hue='Cluster', palette='viridis', s=100)
            # Annotate points with city names
            for i, row in df_clustering.iterrows():
                plt.text(row['Avg_Rent'], row['Avg_Max_Temp'], row['city'], fontsize=9, ha='right')
            plt.title(f'City Clusters based on Housing & Weather Profiles (K={n_clusters})')
            plt.xlabel('Average Rent')
            plt.ylabel('Average Max Temperature (°C)')

            plt.legend(title='Cluster')

        else:

    else:
        print("Not enough cities to perform meaningful clustering (less than 2 cities).")
else:

# --- 4.6 Time Series Forecasting (Average Rent) for all cities ---
print("\n--- 4.6 Time Series Forecasting (Average Rent) for all cities ---")

# Get unique cities from df_combined that have 'VALUE' (rent) data
cities_to_forecast = df_combined['city'].unique()

for forecast_city in cities_to_forecast:
    print(f"\n--- Forecasting Average Rent for {forecast_city} ---")
    df_forecast = df_combined[df_combined['city'] == forecast_city].copy()

    # Aggregate to monthly average rent for the selected city
    df_forecast_monthly = df_forecast.groupby('Date')['VALUE'].mean().reset_index()
    df_forecast_monthly = df_forecast_monthly.rename(columns={'Date': 'ds', 'VALUE': 'y'})
    df_forecast_monthly.set_index('ds', inplace=True)
    df_forecast_monthly.sort_index(inplace=True)

    if not df_forecast_monthly.empty:
        # Split data into training and testing sets for time series (chronological split)
        # Use 80% for training, 20% for testing
        train_size = int(len(df_forecast_monthly) * 0.8)
        train_data = df_forecast_monthly.iloc[:train_size]
        test_data = df_forecast_monthly.iloc[train_size:]

        print(f"  Training data points: {len(train_data)}")
        print(f"  Test data points: {len(test_data)}")

        if len(train_data) < 10: # ARIMA needs a reasonable number of observations
            print(f"  WARNING: Not enough training data for ARIMA ({len(train_data)} points). ARIMA might not perform well or might fail.")
        if len(test_data) == 0:
            print("  WARNING: Test data is empty. Cannot evaluate forecast for this city.")
            continue # Skip to next city if no test data

        # --- ARIMA Model ---
        print("\n  --- ARIMA Model ---")
        try:
            if not train_data.empty:
                arima_model = ARIMA(train_data['y'], order=(5,1,0))
                arima_fit = arima_model.fit()
                # print(arima_fit.summary()) # Uncomment if you want to see full ARIMA summary for each city

                # Make predictions
                arima_predictions = arima_fit.predict(start=len(train_data), end=len(df_forecast_monthly)-1)
                
                # Align predictions with test_data index
                arima_predictions.index = test_data.index

                # Evaluate ARIMA
                rmse_arima = np.sqrt(mean_squared_error(test_data['y'], arima_predictions))
                print(f"  ARIMA RMSE for {forecast_city}: {rmse_arima:.2f}")

                # Plot ARIMA results with visual smoothing for historical data
                plt.figure(figsize=(15, 7))
                
                # Apply spline interpolation for training data if enough points
                # Use datetime.fromordinal for conversion
                if len(train_data) >= 4: # Cubic requires at least 4 points
                    x_train_num = train_data.index.map(pd.Timestamp.toordinal)
                    f_train = interp1d(x_train_num, train_data['y'], kind='cubic')
                    x_train_new = np.linspace(x_train_num.min(), x_train_num.max(), 500)
                    # Convert ordinal back to datetime using datetime.fromordinal
                    smoothed_train_dates = [datetime.fromordinal(int(x)) for x in x_train_new]
                    plt.plot(smoothed_train_dates, f_train(x_train_new), label='Training Data (Smoothed)')
                elif len(train_data) > 1: # Fallback to linear if less than 4 but more than 1
                    x_train_num = train_data.index.map(pd.Timestamp.toordinal)
                    f_train = interp1d(x_train_num, train_data['y'], kind='linear')
                    x_train_new = np.linspace(x_train_num.min(), x_train_num.max(), 500)
                    smoothed_train_dates = [datetime.fromordinal(int(x)) for x in x_train_new]
                    plt.plot(smoothed_train_dates, f_train(x_train_new), label='Training Data (Linear Smoothed)')
                else: # Plot raw if 1 or 0 points
                    plt.plot(train_data.index, train_data['y'], label='Training Data (Raw)')


                # Apply spline interpolation for actual test data if enough points
                # Use datetime.fromordinal for conversion
                if len(test_data) >= 4: # Cubic requires at least 4 points
                    x_test_num = test_data.index.map(pd.Timestamp.toordinal)
                    f_test = interp1d(x_test_num, test_data['y'], kind='cubic')
                    x_test_new = np.linspace(x_test_num.min(), x_test_num.max(), 100)
                    smoothed_test_dates = [datetime.fromordinal(int(x)) for x in x_test_new]
                    plt.plot(smoothed_test_dates, f_test(x_test_new), label='Actual Test Data (Smoothed)', color='orange')
                elif len(test_data) > 1: # Fallback to linear if less than 4 but more than 1
                    x_test_num = test_data.index.map(pd.Timestamp.toordinal)
                    f_test = interp1d(x_test_num, test_data['y'], kind='linear')
                    x_test_new = np.linspace(x_test_num.min(), x_test_num.max(), 100)
                    smoothed_test_dates = [datetime.fromordinal(int(x)) for x in x_test_new]
                    plt.plot(smoothed_test_dates, f_test(x_test_new), label='Actual Test Data (Linear Smoothed)', color='orange')
                else: # Plot raw if 1 or 0 points
                    plt.plot(test_data.index, test_data['y'], label='Actual Test Data (Raw)', color='orange')

                plt.plot(arima_predictions.index, arima_predictions, label='ARIMA Predictions', color='green', linestyle='--')
                plt.title(f'ARIMA Forecast for Average Rent in {forecast_city} (Visually Smoothed Historical Data)')
                plt.xlabel('Date')
                plt.ylabel('Average Rent')
                plt.legend()
                plt.grid(True, linestyle='--', alpha=0.6)
                plt.tight_layout()
                plt.show()
            else:
                print(f"  ARIMA skipped for {forecast_city}: Training data is empty.")

        except Exception as e:
            print(f"  Error training or predicting with ARIMA for {forecast_city}: {e}")
            print("  ARIMA might fail if data is too short, not stationary, or not suitable for the chosen order.")


        # --- Prophet Model ---
        print("\n  --- Prophet Model ---")
        prophet_df = df_forecast_monthly.reset_index()
        train_prophet = prophet_df.iloc[:train_size]
        test_prophet = prophet_df.iloc[train_size:]

        try:
            if not train_prophet.empty:
                prophet_model = Prophet(
                    seasonality_mode='multiplicative',
                    changepoint_prior_scale=0.05
                )
                prophet_model.add_country_holidays(country_name='CA') # Add Canadian holidays
                prophet_model.fit(train_prophet)

                future = prophet_model.make_future_dataframe(periods=len(test_data), freq='MS')
                forecast = prophet_model.predict(future)

                prophet_predictions = forecast['yhat'].iloc[train_size:]
                prophet_predictions.index = test_data.index

                rmse_prophet = np.sqrt(mean_squared_error(test_data['y'], prophet_predictions))
                print(f"  Prophet RMSE for {forecast_city}: {rmse_prophet:.2f}")

                # Plot Prophet results with visual smoothing for historical data
                plt.figure(figsize=(15, 7))
                
                # Apply spline interpolation for training data if enough points
                if len(train_data) >= 4: # Cubic requires at least 4 points
                    x_train_num = train_data.index.map(pd.Timestamp.toordinal)
                    f_train = interp1d(x_train_num, train_data['y'], kind='cubic')
                    x_train_new = np.linspace(x_train_num.min(), x_train_num.max(), 500)
                    smoothed_train_dates = [datetime.fromordinal(int(x)) for x in x_train_new]
                    plt.plot(smoothed_train_dates, f_train(x_train_new), label='Training Data (Smoothed)')
                elif len(train_data) > 1: # Fallback to linear if less than 4 but more than 1
                    x_train_num = train_data.index.map(pd.Timestamp.toordinal)
                    f_train = interp1d(x_train_num, train_data['y'], kind='linear')
                    x_train_new = np.linspace(x_train_num.min(), x_train_num.max(), 500)
                    smoothed_train_dates = [datetime.fromordinal(int(x)) for x in x_train_new]
                    plt.plot(smoothed_train_dates, f_train(x_train_new), label='Training Data (Linear Smoothed)')
                else: # Plot raw if 1 or 0 points
                    plt.plot(train_data.index, train_data['y'], label='Training Data (Raw)')


                # Apply spline interpolation for actual test data if enough points
                if len(test_data) >= 4: # Cubic requires at least 4 points
                    x_test_num = test_data.index.map(pd.Timestamp.toordinal)
                    f_test = interp1d(x_test_num, test_data['y'], kind='cubic')
                    x_test_new = np.linspace(x_test_num.min(), x_test_num.max(), 100)
                    smoothed_test_dates = [datetime.fromordinal(int(x)) for x in x_test_new]
                    plt.plot(smoothed_test_dates, f_test(x_test_new), label='Actual Test Data (Smoothed)', color='orange')
                elif len(test_data) > 1: # Fallback to linear if less than 4 but more than 1
                    x_test_num = test_data.index.map(pd.Timestamp.toordinal)
                    f_test = interp1d(x_test_num, test_data['y'], kind='linear')
                    x_test_new = np.linspace(x_test_num.min(), x_test_num.max(), 100)
                    smoothed_test_dates = [datetime.fromordinal(int(x)) for x in x_test_new]
                    plt.plot(smoothed_test_dates, f_test(x_test_new), label='Actual Test Data (Linear Smoothed)', color='orange')
                else: # Plot raw if 1 or 0 points
                    plt.plot(test_data.index, test_data['y'], label='Actual Test Data (Raw)', color='orange')


                plt.plot(prophet_predictions.index, prophet_predictions, label='Prophet Predictions', color='purple', linestyle='--')
                plt.title(f'Prophet Forecast for Average Rent in {forecast_city} (Visually Smoothed Historical Data)')
                plt.xlabel('Date')
                plt.ylabel('Average Rent')
                plt.legend()
                plt.grid(True, linestyle='--', alpha=0.6)
                plt.tight_layout()
                plt.show()

            else:
                print(f"  Prophet skipped for {forecast_city}: Training data is empty.")

        except Exception as e:
            print(f"  Error training or predicting with Prophet for {forecast_city}: {e}")
            print("  Prophet might require more data or specific data characteristics.")

    else:
        print(f"No sufficient monthly average rent data for {forecast_city} to perform time series forecasting.")
